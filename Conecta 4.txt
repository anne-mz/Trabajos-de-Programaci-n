#include <stdio.h>
#include <unistd.h>

int tamañotablerocolumnas = 7;
int tamañotablerofilas = 8;
   
char tablero[8][7];
int jugador = 1;
   
    void inicializartablero(){
       for  (int i = 0; i < tamañotablerofilas; i++){
           for(int j = 0; j < tamañotablerocolumnas; j++){
               tablero[i][j] = ' ';
           }
       }
   }

void mostrartablero(){
    printf("\e[1;1H\e[2J");
    
    for(int i = 0; i < tamañotablerofilas; i++){
        for(int j = 0; j < tamañotablerocolumnas; j++){
            printf("|%c", tablero[i][j]);
        }
        printf("|\n");
    }
}

///poner fichas en el tablero de ambos jugadores
///jugador 1 X 
///jugador 2 O
void ponerficha(){
    int columna;
    char fichaActual = (jugador == 1) ? 'X' : 'O';
    
    printf("jugador %d, ingrese la columna (0-6) para colocar la ficha: ", jugador);
    scanf("%d", &columna);
    
    ///Validar columna
    if(columna < 0 || columna >= tamañotablerocolumnas){
        printf("Columna invalida paps. intenta again. \n");
        return;
    }
    
    ///Buscar fila destino
    int filadestino = -1;
    for(int i = tamañotablerofilas - 1; i >= 0; i--){
        if(tablero[i][columna] == ' '){
            filadestino = i;
            break;
        }
    }
    
    ///Si la columna esta llena
    if(filadestino == -1){
        printf("¡La columna está llena! Intenta en otra. \n");
        return;
    }
    
    ///Animación de caida
    for(int i = 0; i <= filadestino; i++){
        
        tablero[i][columna] = fichaActual;
        
        mostrartablero();
        
        usleep(100000);
        
        if(i < filadestino) {
            tablero[i][columna] = ' ';
        }
    }
    
    ///Cambiar turno
    if(jugador == 1){
        jugador = 2;
        }else{
            jugador = 1;
    }

}

int verificarganador(char ficha){
    
    /////1.Verificación Horizontal
    for (int i = 0; i < tamañotablerofilas; i++){
        for(int j = 0; j < tamañotablerocolumnas - 3; j++){
            if (tablero[i][j] == ficha &&
            tablero[i][j + 1] == ficha &&
            tablero[i][j + 2] == ficha &&
            tablero[i][j + 3] == ficha) {
                return 1;
            }
        }
    }
    
    /////2.Verificacion Vertical
    for (int i = 0; i < tamañotablerofilas - 3; i++){
        for(int j = 0; j < tamañotablerocolumnas; j++){
            if (tablero[i][j] == ficha &&
            tablero[i + 1][j] == ficha &&
            tablero[i + 2][j] == ficha &&
            tablero[i + 3][j] == ficha) {
                return 1;
            }
        }
    }
    
    /////3.Verifiacion Diagonal Descendente
    for (int i = 0; i < tamañotablerofilas -3; i++){
        for(int j = 0; j < tamañotablerocolumnas - 3; j++){
            if (tablero[i][j] == ficha &&
            tablero[i + 1][j + 1] == ficha &&
            tablero[i + 2][j + 2] == ficha &&
            tablero[i + 3][j + 3] == ficha) {
                return 1;
            }
        }
    }
    
    /////4.Verificacion Diagonal Ascendente
    for (int i = 3; i < tamañotablerofilas; i++){
        for(int j = 0; j < tamañotablerocolumnas - 3; j++){
            if (tablero[i][j] == ficha &&
            tablero[i - 1][j + 1] == ficha &&
            tablero[i - 2][j + 2] == ficha &&
            tablero[i - 3][j + 3] == ficha) {
                return 1;
            }
        }
    }
}
 int main(){
     inicializartablero();
     
     int turnostotales = tamañotablerofilas * tamañotablerocolumnas;
     int juegoterminado = 0;
     
     mostrartablero();
     
     for(int i = 0; i < turnostotales && !juegoterminado; i++){
         
         ponerficha();
         
         if (verificarganador('X')){
             printf("\n Ganó el jugador 1\n");
             juegoterminado = 1;
         }
         
         else if(verificarganador('O')){
             printf("\n Ganó el jugador 2\n");
             juegoterminado = 1;
         }
     }
     
     if(!juegoterminado){
         printf("\n Empataron pá\n");
     }
     
     
    return 0;
}
