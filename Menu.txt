#include <stdio.h>

int main()
{
   
   int menu = 1;
    
    do{
        
        printf("Qué quieres hacer en el menu \n");
        printf("1.- calcular mi promedio de calificación \n");
        printf("2.- adios mundo \n");
        printf("3.- salir \n");
        scanf("%d", &menu);
        
        switch(menu){
            case 1:
             int cantidadmaterias;
             float calificacion, promedio;
             
             printf("ingresa la cantidad de materias: ");
             scanf("%d", &cantidadmaterias);
             
             for(int i = 0; i < cantidadmaterias; i++){
                 printf("ingresa calificación %d:", i + 1);
                 scanf("%f", &calificacion);
                 promedio += calificacion;
             }
             
             promedio /= cantidadmaterias;
             printf("el promedio es %.2f \n", promedio);
             
             
             break;
            case 2:
             printf("adios mundo \n");
             break;
            case 3:
             printf("saliste del menu \n");
             break;
            default:
             printf("opción no valida \n");
        }
        
        
        
        
        
    }while (menu != 3);
    
    return 0;
}
