#include <stdio.h>

int main(){
    
    char inicial;
    float numero = 7;
    float calificacion1, calificacion2, calificacion3, calificacion4, calificacion5, calificacionfinal;
    
    printf("¿Cúal es la inicial de tu nombre? \n");
    scanf("%c", &inicial);
    
    printf("Qué calificación sacaste en el primer parcial?  ");
    scanf("%f", &calificacion1);
    
    printf("Qué calificación sacaste en el segundo parcial? ");
    scanf("%f", &calificacion2);
    
    printf("Qué calificación sacaste en el tercer parcial? ");
    scanf("%f", &calificacion3);
    
    printf("Qué calificación sacaste en el cuarto parcial? ");
    scanf("%f", &calificacion4);
    
    printf("Qué calificación sacaste en el quinto parcial? ");
    scanf("%f", &calificacion5);
    
    calificacionfinal = (calificacion1 + calificacion2 + calificacion3 + calificacion4 + calificacion5) / 5;
    
    printf("Tu inicial es %c calificación final es de %.2f \n" , inicial, calificacionfinal);
    
    if(calificacionfinal > numero){
    printf("Felicidades, pasaste!! \n");
    }else{
    printf("no, no pasaste :( \n");    
    }
    
    
    return 0;
}
