#include <stdio.h>
#include <math.h>

int main() {
    float cata, catb, hipotenusa;

    printf("Cúal el valor del primer cateto? ");
    scanf("%f", &cata);

    printf("Cúal el valor del segundo cateto? ");
    scanf("%f", &catb);

    hipotenusa = sqrt((cata * cata) + (catb * catb));

    printf("La hipotenusa es: %.2f \n", hipotenusa);

    return 0;
}
