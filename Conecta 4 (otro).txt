#include <stdio.h>
#include <stdlib.h>

#define FILAS 6
#define COLUMNAS 7

void inicializarTablero(char tablero[FILAS][COLUMNAS]) {
    for (int i = 0; i < FILAS; i++) {
        for (int j = 0; j < COLUMNAS; j++) {
            tablero[i][j] = '.';
        }
    }
}

void imprimirTablero(char tablero[FILAS][COLUMNAS]) {
    printf("\n 0 1 2 3 4 5 6\n");
    for (int i = 0; i < FILAS; i++) {
        for (int j = 0; j < COLUMNAS; j++) {
            printf(" %c", tablero[i][j]);
        }
        printf("\n");
    }
    printf("---------------\n");
}

int colocarFicha(char tablero[FILAS][COLUMNAS], int columna, char ficha) {
    if (columna < 0 || columna >= COLUMNAS) return 0;
    for (int i = FILAS - 1; i >= 0; i--) {
        if (tablero[i][columna] == '.') {
            tablero[i][columna] = ficha;
            return 1;
        }
    }
    return 0; // Columna llena
}

int verificarGanador(char tablero[FILAS][COLUMNAS], char ficha) {
    // Horizontal
    for (int i = 0; i < FILAS; i++) {
        for (int j = 0; j < COLUMNAS - 3; j++) {
            if (tablero[i][j] == ficha && tablero[i][j+1] == ficha &&
                tablero[i][j+2] == ficha && tablero[i][j+3] == ficha) return 1;
        }
    }
    // Vertical
    for (int i = 0; i < FILAS - 3; i++) {
        for (int j = 0; j < COLUMNAS; j++) {
            if (tablero[i][j] == ficha && tablero[i+1][j] == ficha &&
                tablero[i+2][j] == ficha && tablero[i+3][j] == ficha) return 1;
        }
    }
    // Diagonal descendente (\)
    for (int i = 0; i < FILAS - 3; i++) {
        for (int j = 0; j < COLUMNAS - 3; j++) {
            if (tablero[i][j] == ficha && tablero[i+1][j+1] == ficha &&
                tablero[i+2][j+2] == ficha && tablero[i+3][j+3] == ficha) return 1;
        }
    }
    // Diagonal ascendente (/)
    for (int i = 3; i < FILAS; i++) {
        for (int j = 0; j < COLUMNAS - 3; j++) {
            if (tablero[i][j] == ficha && tablero[i-1][j+1] == ficha &&
                tablero[i-2][j+2] == ficha && tablero[i-3][j+3] == ficha) return 1;
        }
    }
    return 0;
}

int main() {
    char tablero[FILAS][COLUMNAS];
    inicializarTablero(tablero);
    char fichas[2] = {'X', 'O'};
    int turno = 0;
    int movimientos = 0;

    while (movimientos < FILAS * COLUMNAS) {
        imprimirTablero(tablero);
        char fichaActual = fichas[turno % 2];
        printf("Turno del jugador %c. Elige columna (0-6): ", fichaActual);
        
        int col;
        if (scanf("%d", &col) != 1) {
            while (getchar() != '\n'); // limpiar buffer
            continue;
        }

        if (colocarFicha(tablero, col, fichaActual)) {
            if (verificarGanador(tablero, fichaActual)) {
                imprimirTablero(tablero);
                printf("¡El jugador %c ha ganado!\n", fichaActual);
                return 0;
            }
            turno++;
            movimientos++;
        } else {
            printf("Movimiento inválido. Intenta de nuevo.\n");
        }
    }

    imprimirTablero(tablero);
    printf("¡Empate!\n");
    return 0;
}
